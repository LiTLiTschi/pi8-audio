# Display v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add online cover art lookup (SoundCloud → Deezer → iTunes → gradient fallback), structured logging, and a responsive Web UI (tabs on mobile, two-column on desktop) to the audio-sync-web + display-daemon stack.

**Architecture:** `display-daemon` gains a `get_cover_art()` pipeline that tries three APIs in order before falling back to a deterministic gradient. `audio-sync-web` gains a CSS tab/two-column layout wrapping the existing panels — no server-side changes needed.

**Tech Stack:** Python 3, Pillow, numpy, `urllib.request` (stdlib only — no new pip deps), systemd user services, mpv DRM, BlueZ AVRCP via mpris-proxy.

```python
# ── Online cover art lookup ────────────────────────────────────────────────────
_art_cache: dict = {}   # key: (artist, title) → url or ""

def get_cover_art(artist: str, title: str, album: str = "") -> str:
    """Fetch cover art URL. Tries SoundCloud → Deezer → iTunes. Returns "" on failure."""
    key = (artist.lower(), title.lower())
    if key in _art_cache:
        return _art_cache[key]

    url = (_sc_cover(artist, title)
           or _deezer_cover(artist, title)
           or _itunes_cover(artist, title))

    _art_cache[key] = url or ""
    if url:
        log.info("cover art found for %r — %r", artist, title)
    else:
        log.info("no cover art found for %r — %r (will use gradient)", artist, title)
    return _art_cache[key]


def _sc_cover(artist: str, title: str) -> str:
    client_id = _display_config.get("soundcloud_client_id", "")
    if not client_id:
        return ""
    try:
        q = urllib.parse.quote(f"{artist} {title}")
        req = urllib.request.Request(
            f"https://api.soundcloud.com/tracks?q={q}&client_id={client_id}&limit=3",
            headers={"User-Agent": "display-daemon/2.0"})
        data = json.loads(urllib.request.urlopen(req, timeout=5).read())
        for track in data:
            art = track.get("artwork_url") or track.get("user", {}).get("avatar_url", "")
            if art:
                return art.replace("large", "t500x500")
    except Exception as e:
        log.debug("SoundCloud lookup failed: %s", e)
    return ""


def _deezer_cover(artist: str, title: str) -> str:
    try:
        q = urllib.parse.quote(f"{artist} {title}")
        req = urllib.request.Request(
            f"https://api.deezer.com/search?q={q}&limit=5",
            headers={"User-Agent": "display-daemon/2.0"})
        data = json.loads(urllib.request.urlopen(req, timeout=5).read())
        for item in data.get("data", []):
            album = item.get("album", {})
            url = album.get("cover_xl") or album.get("cover_big") or album.get("cover_medium")
            if url:
                return url
    except Exception as e:
        log.debug("Deezer lookup failed: %s", e)
    return ""


def _itunes_cover(artist: str, title: str) -> str:
    try:
        q = urllib.parse.quote(f"{artist} {title}")
        req = urllib.request.Request(
            f"https://itunes.apple.com/search?term={q}&entity=musicTrack&limit=1",
            headers={"User-Agent": "display-daemon/2.0"})
        data = json.loads(urllib.request.urlopen(req, timeout=5).read())
        results = data.get("results", [])
        if results:
            art = results[0].get("artworkUrl100", "")
            if art:
                return art.replace("100x100bb.jpg", "600x600bb.jpg")
    except Exception as e:
        log.debug("iTunes lookup failed: %s", e)
    return ""


def gradient_from_artist(artist: str, W: int, H: int) -> "Image.Image":
    """Generate a deterministic vertical gradient from artist name hash."""
    import numpy as np
    h = int(hashlib.md5(artist.encode() or b"unknown").hexdigest()[:8], 16)
    hue1 = (h & 0xFF) / 255.0
    hue2 = ((h >> 8) & 0xFF) / 255.0
    rgb1 = tuple(int(x * 255) for x in colorsys.hsv_to_rgb(hue1, 0.65, 0.30))
    rgb2 = tuple(int(x * 255) for x in colorsys.hsv_to_rgb(hue2, 0.75, 0.50))
    t = np.linspace(0, 1, H, dtype=np.float32)[:, None]
    arr = (np.array(rgb1, dtype=np.float32) * (1 - t)
           + np.array(rgb2, dtype=np.float32) * t)
    arr = np.broadcast_to(arr[:, None, :], (H, W, 3)).astype(np.uint8).copy()
    return Image.fromarray(arr, "RGB")
```

## Edge Cases & Notes

| Scenario | Handling |
|---|---|
| SoundCloud client_id empty | `_sc_cover` returns `""` immediately, Deezer tried next |
| All APIs fail / no network | `get_cover_art` returns `""`, gradient rendered |
| Same artist/track plays again | `_art_cache` hit, no HTTP request |
| artUrl present in MPRIS (rare) | Used directly, online lookup skipped |
| Cover art download fails | Warning logged, gradient shown |
| Track with no artist AND no title | Gradient uses `""` hash — consistent purple-ish color |
| Desktop browser < 900px | Tab layout used (responsive) |
| `display-config.json` missing | `load_display_config()` returns `{}`, SoundCloud silently skipped |
